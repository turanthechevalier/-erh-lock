import os
import json
from flask import Flask, request, jsonify
from flask_cors import CORS
import google.generativeai as genai
from supabase import create_client, Client

app = Flask(__name__)
CORS(app) 

# Çevre Değişkenleri (Render'da tanımlanacak)
SUPABASE_URL = os.environ.get("SUPABASE_URL")
SUPABASE_KEY = os.environ.get("SUPABASE_SERVICE_ROLE_KEY")
GEMINI_API_KEY = os.environ.get("GEMINI_API_KEY")

supabase: Client = create_client(SUPABASE_URL, SUPABASE_KEY)
genai.configure(api_key=GEMINI_API_KEY)
model = genai.GenerativeModel('gemini-2.5-flash')

@app.route('/serh-isle', methods=['POST'])
def serh_isle():
    data = request.json
    
    prompt = f"""
    Sen anayasa hukuku ve genel kamu hukuku alanında uzman bir akademik platform editörüsün. Görevin, aşağıdaki ana metne düşülen kullanıcı şerhini incelemektir. 
    Argümanın kavramsal tutarlılığını (örneğin egemenlik, teknofeodalizm, istisnai hal gibi kavramların doğru kullanımını) fact-check et.
    
    Ana Metindeki Parça: "{data.get('hedef_metin')}"
    Kullanıcının Şerhi: "{data.get('kullanici_serhi')}"
    
    Çıktı formatın sadece aşağıdaki JSON yapısı olmalıdır, başka hiçbir kelime yazma:
    {{
        "analiz": "Kısa akademik değerlendirme",
        "karar": "Onaylandı" veya "Kısmi Kabul" veya "Reddedildi",
        "puan_etkisi": 20 (Onay), 5 (Kısmi), -15 (Red)
    }}
    """
    
    try:
        response = model.generate_content(prompt)
        ai_cikti = response.text.strip().replace('```json', '').replace('```', '').strip()
        ai_sonuc = json.loads(ai_cikti)

        # Şerhi Veritabanına Yaz
        supabase.table('serhler').insert({
            "metin_id": data.get('metin_id'),
            "yazar_id": data.get('yazar_id'),
            "hedef_metin": data.get('hedef_metin'),
            "kullanici_serhi": data.get('kullanici_serhi'),
            "ai_analiz_raporu": ai_sonuc['analiz'],
            "ai_karari": ai_sonuc['karar'],
            "puan_degisimi": ai_sonuc['puan_etkisi']
        }).execute()

        # Ponzi / İtibar Puanını Güncelle
        kullanici = supabase.table('proprofiles').select('itibar_puani').eq('id', data.get('yazar_id')).single().execute()
        # Not: Yukarıdaki test adımında oluşturduğumuz dummy kullanıcı ile deneme yapacağız.
        
        return jsonify({
            "basari": True,
            "ai_karari": ai_sonuc['karar'],
            "analiz": ai_sonuc['analiz'],
            "puan": ai_sonuc['puan_etkisi']
        })

    except Exception as e:
        return jsonify({"basari": False, "hata": str(e)}), 500

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)
